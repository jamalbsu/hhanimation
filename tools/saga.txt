// ---- SAGA: the star the pack carries, the words they trade, and a deeper picture and sound ----
const hex3 = h => [parseInt(h.slice(1, 3), 16), parseInt(h.slice(3, 5), 16), parseInt(h.slice(5, 7), 16)];
function colAt(keys, t) {
  if (t <= keys[0][0]) return hex3(keys[0][1]);
  for (let i = 0; i < keys.length - 1; i++) if (t <= keys[i + 1][0]) { const k = smooth(clamp((t - keys[i][0]) / (keys[i + 1][0] - keys[i][0]))), p = hex3(keys[i][1]), q = hex3(keys[i + 1][1]); return [lerp(p[0], q[0], k), lerp(p[1], q[1], k), lerp(p[2], q[2], k)]; }
  return hex3(keys[keys.length - 1][1]);
}
// the star: it falls, waits on the ground, rides above the pack trailing a ribbon of light, and at last rises home
const STAR = CFG.star ? { ...CFG.star, D: tab(TAB0.d, CFG.star.from) + 110 } : null;
function starPos(t) {
  if (STL < STAR.from) return { f: STAR.D - SD, hg: 60 + 14 * Math.sin(t * 4), held: 0, up: 0 };
  const up = STAR.release !== undefined && STL > STAR.release ? STL - STAR.release : 0, k = smooth(clamp((STL - STAR.from) / 0.5));
  return { f: lerp(110, 50, k), hg: lerp(60, 330, k) + 18 * Math.sin(t * 3.1) + up * up * 900, held: 1, up };
}
function drawStar(t) {
  if (!STAR) return; ctx.setTransform(1, 0, 0, 1, 0, 0);
  const c = colAt(STAR.col, STL), cs = c.map(v => v | 0).join(',');
  if (STAR.fall !== undefined && STL < STAR.fall) {                  // the meteor streaks down to where it will land
    const s = clamp((STL - (STAR.fall - 1.25)) / 1.25); if (s <= 0 || !hzVis(STAR.D - SD, 0)) return;
    const land = P3s(STAR.D - SD, 60, 0), x0 = W * 1.05, y0 = -H * 0.08, e = s * s, hx = lerp(x0, land[0], e), hy = lerp(y0, land[1], e), tx = lerp(x0, land[0], Math.max(0, e - 0.3)), ty = lerp(y0, land[1], Math.max(0, e - 0.3));
    ctx.globalCompositeOperation = 'lighter'; ctx.lineCap = 'round';
    const g = ctx.createLinearGradient(tx, ty, hx, hy); g.addColorStop(0, `rgba(${cs},0)`); g.addColorStop(1, `rgba(${cs},0.9)`);
    ctx.strokeStyle = g; ctx.lineWidth = S * 0.02; ctx.beginPath(); ctx.moveTo(tx, ty); ctx.lineTo(hx, hy); ctx.stroke();
    ctx.strokeStyle = 'rgba(255,255,255,0.9)'; ctx.lineWidth = S * 0.006; ctx.beginPath(); ctx.moveTo(lerp(tx, hx, 0.5), lerp(ty, hy, 0.5)); ctx.lineTo(hx, hy); ctx.stroke();
    hzPuff(hx, hy, S * 0.07, 'rgba(255,255,255,0.95)', `rgba(${cs},0.5)`, `rgba(${cs},0)`);
    ctx.globalCompositeOperation = 'source-over'; return;
  }
  const p0 = starPos(t), fade = clamp(1 - p0.up / 1.3);
  ctx.globalCompositeOperation = 'lighter'; ctx.lineCap = 'round';
  if (p0.held && fade > 0) {                                           // the ribbon of light streaming behind
    const N = 46, pts = [];
    for (let i = 0; i <= N; i++) { const a = i / N, f = p0.f - i * 34, z = 110 * a * Math.sin(t * 1.9 - i * 0.21), hg = p0.hg * (1 - 0.35 * a) + 55 * a * Math.sin(t * 3.2 - i * 0.33); pts.push(hzVis(f, z) ? [P3s(f, hg, z), hzK(f, hg, z), a] : null); }
    for (const [w, al, col] of [[30, 0.14, cs], [11, 0.5, cs], [3.2, 0.85, '255,255,255']]) for (let i = 0; i < N; i++) {
      const p = pts[i], q = pts[i + 1]; if (!p || !q) continue;
      ctx.strokeStyle = `rgba(${col},${al * fade * (1 - p[2]) })`; ctx.lineWidth = Math.max(0.6, w * p[1] * (1 - 0.8 * p[2])); ctx.beginPath(); ctx.moveTo(p[0][0], p[0][1]); ctx.lineTo(q[0][0], q[0][1]); ctx.stroke();
    }
    ctx.fillStyle = `rgba(255,255,255,${0.9 * fade})`;
    for (let i = 3; i < N; i += 4) { const p = pts[i]; if (!p) continue; const tw = Math.max(0, Math.sin(t * 6 + i * 1.9)); if (tw > 0.2) sparkle(p[0][0] + 14 * p[1] * Math.sin(i * 2.7), p[0][1] - 18 * p[1] * Math.cos(i * 1.3), 9 * p[1] * tw * (1 - p[2])); }
  }
  if (hzVis(p0.f, 0)) {                                                // the star itself
    const p = P3s(p0.f, p0.hg, 0), k = hzK(p0.f, p0.hg, 0), pulse = 1 + 0.12 * Math.sin(t * 7);
    hzPuff(p[0], p[1], 120 * k * pulse, `rgba(${cs},0.7)`, `rgba(${cs},0.25)`, `rgba(${cs},0)`);
    ctx.fillStyle = `rgba(${cs},0.95)`; ctx.save(); ctx.translate(p[0], p[1]); ctx.rotate(t * 0.9); sparkle(0, 0, 46 * k * pulse); ctx.rotate(Math.PI / 4); sparkle(0, 0, 26 * k * pulse); ctx.restore();
    ctx.fillStyle = '#ffffff'; ctx.beginPath(); ctx.arc(p[0], p[1], 9 * k, 0, TAU); ctx.fill();
  }
  ctx.globalCompositeOperation = 'source-over';
}

    const d = STL - g.t0, life = g.t1 - g.t0; if (d < 0 || d > life) continue;

    const pin = backOut(clamp(d / 0.22)), out = 1 - smooth(clamp((d - life + 0.25) / 0.25)), fs = S * (g.size || 0.044);
    ctx.font = `700 ${Math.round(fs)}px "Comic Sans MS", "Segoe UI", sans-serif`;
    const lines = g.text.split('|'), tw = Math.max(...lines.map(l => ctx.measureText(l).width)), bw = tw + fs * 1.5, bh = lines.length * fs * 1.25 + fs * 0.85;
    const ax = clamp(A[0], S * 0.08, W - S * 0.08), bx = clamp(ax - bw * (g.side === -1 ? 0.75 : 0.25), S * 0.03, W - bw - S * 0.03), by = clamp(A[1] - S * 0.11 - bh, H * 0.07, H * 0.82 - bh);
    const cx = bx + bw / 2, cy = by + bh, tx = clamp(ax, bx + fs * 1.1, bx + bw - fs * 1.1), tip = [lerp(tx, A[0], 0.8), lerp(cy, A[1], 0.8)];
    ctx.save(); ctx.globalAlpha = out; ctx.translate(cx, cy); ctx.scale(pin, pin); ctx.translate(-cx, -cy); ctx.lineJoin = 'round';
    ctx.fillStyle = INK; ctx.beginPath(); ctx.roundRect(bx + S * 0.009, by + S * 0.011, bw, bh, fs * 0.7); ctx.fill();
