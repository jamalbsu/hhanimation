function legAng(p, hind, amp, A) {              // upper-leg swing (+ = forward) and the bend below the knee or hock
  const duty = lerp(0.6, 0.36, amp), Aa = A * lerp(0.3, 0.62, amp);
  if (p < duty) return [lerp(Aa, -Aa, p / duty), 0];
  const s = (p - duty) / (1 - duty), up = lerp(-Aa, Aa, smooth(s)), fold = Math.sin(Math.PI * s) * lerp(0.6, 1.5, amp) * clamp(A * 2);
  return [up, hind ? fold * 0.8 : -fold];
}

    if ((A.birds || 0) > 0.05 && hh(i, 3.1) < 0.12 * A.birds) EVENTS.push({ t, k: 'tweet', v: A.birds, f: 2600 + 1900 * hh(i, 4.2), pan: (hh(i, 5.3) - 0.5) * 1.4 });

  }
  for (let t = 0.1; t < STORY - 0.3; t += 0.09) for (const h of HERDS) {        // hooves: a drumming that swells as the herd comes close
    let near = 1e9, n = 0; const v = tab(h.tb.v, t); if (v < 60) continue;
    h.members.forEach((m, mi) => { const d = Math.hypot(herdFx(h, m, t), herdZ(h, m, mi, t) * 0.6); near = Math.min(near, d); if (d < 2500) n++; });
    const vol = Math.sqrt(n) * 0.075 * clamp(1 - near / 2600) * clamp(v / 300) * (h.kind === 'buffalo' || h.kind === 'rhino' ? 1.5 : 0.8);
    if (vol > 0.004) EVENTS.push({ t: t + 0.03 * Math.sin(t * 50), k: 'hoof', v: vol, heavy: h.sp.L > 100, pan: (hh(t * 10 | 0, 2) - 0.5) * 0.8 });
  }
  for (const h of HERDS) for (const t of h.snorts || []) EVENTS.push({ t, k: 'snort', heavy: h.sp.L > 100 });
  for (const t of CFG.roars || []) EVENTS.push({ t, k: 'roar' });
  for (const [a, b, bpm] of CFG.drums || []) for (let t = a, i = 0; t < b; t += 60 / bpm, i++) EVENTS.push({ t, k: 'drum', acc: i % 4 === 0, v: clamp((t - a) / 1.5) * clamp((b - t) / 1.5) });
})();

let AX = 0, AY = 0, AK = 1;   // where the animal being drawn sits relative to the panther (screen body units), and its size
function placeAnimal(fx, z, size) {
  const o = pr(0, 0, 0), p = pr(fx, 0, z);
  AX = p[0] - o[0]; AY = p[1] - o[1];
  AK = size * camK(fx, PIV, z) / camK(0, PIV, 0);
}
const tooClose = (fx, z) => CH_D + (fx * RS - z * RC) * CT < CH_D * 0.42;
function drawPack(t) {
  let pb = 0;
  for (const h of HERDS) if (h.spread) h.members.forEach(m => { pb = Math.max(pb, smooth(clamp(1 - Math.abs(herdFx(h, m, STL)) / 700))); });
  const spread = 1 + 0.35 * pb, tau = SD / 450;
  const ents = [{ fx: 0, z: 0, me: true }];
  OURS.forEach((a, i) => ents.push({ a, fa: 1, ph: a.ph, fx: a.fx + 40 * Math.sin(tau * 0.8 + i * 2.1), z: a.z * spread + 10 * Math.sin(tau * 0.6 + i) }));
  for (const h of HERDS) {
    const Dm = tab(h.tb.d, STL), vm = tab(h.tb.v, STL), hdK = h.hd ? keyVal(h.hd, STL) : 0;
    h.members.forEach((m, mi) => {
      const fx = herdFx(h, m, STL); if (fx < -4500 || fx > 6500) return;
      ents.push({ beast: true, h, m, mi, fx, z: herdZ(h, m, mi, STL), Dm, vm, hdK });
    });
  }
  for (const e of ents) e.d = e.fx * RS - e.z * RC;              // further from the camera first
  ents.sort((p, q) => q.d - p.d);
  for (const e of ents) {
    TALKW = 0;
    if (e.me) { FA = 1; AX = 0; AY = 0; AK = 1; COAT = COATS.panther; MARKS = null; DLAG = 0; TAB = TAB0; drawPanther(t); continue; }
    if (tooClose(e.fx, e.z)) continue;
    if (e.beast) {
      const { h, m, mi } = e, sp = h.sp, size = m.size || 1;
      FA = 1; placeAnimal(e.fx, e.z, size); if (AK > 2.6 || Math.abs(AX) > 1600) continue;
      let lift = 0, leap = 0, lp = 0;
      for (const [l0, ld, lh] of h.leaps || []) { const s = (STL - l0 - mi * 0.13) / ld; if (s > 0 && s < 1) { lift = Math.max(lift, lh * Math.sin(Math.PI * s)); leap = Math.max(leap, Math.sin(Math.PI * s) * 1.4); lp = -0.18 * Math.cos(Math.PI * s); } }
      const amp = clamp((e.vm - 150) / 300), cyc = e.Dm / (sp.stride * size) + (m.ph === undefined ? hh(mi, 4.4) : m.ph);
      const graze = e.vm < 20 ? 6 * Math.sin(STL * 1.3 + mi * 2) : 0;
      const st = { cyc, amp, A: clamp(e.vm / 90), lift: lift + amp * 8 * Math.max(0, Math.sin(TAU * (cyc - 0.25))), pitch: amp * 0.06 * Math.sin(TAU * cyc) + lp, hd: e.hdK + graze, leap: clamp(leap), tail: STL * (3 + 6 * amp) + mi };
      FA = h.dir; setGroundTransform(ctx); beastShadow(sp, st.lift); drawBeast(sp, st);
      ctx.setTransform(1, 0, 0, 1, 0, 0);
      continue;
    }
    FA = 1; placeAnimal(e.fx, e.z, e.a.size);
    if (AK > 2.4 || Math.abs(AX) > 1400) continue;
    FA = e.fa; COAT = e.a.coat; MARKS = e.a.marks; DLAG = 0; TAB = TAB0;
    drawPanther(t, e.ph);
  }
  FA = 1; AX = 0; AY = 0; AK = 1; COAT = COATS.panther; MARKS = null; DLAG = 0; TALKW = 0; TAB = TAB0;
}
// dust thrown up behind the running herds
function drawHerdDust() {
  ctx.setTransform(1, 0, 0, 1, 0, 0);
  const P3w = (f, z, hg) => { const q = pr(f, -hg, z); return [ox + CAMX + q[0] * sc, oy + CAMY + q[1] * sc]; };
  for (const h of HERDS) {
    if (!h.dust) continue;
    const Dm = tab(h.tb.d, STL), vm = tab(h.tb.v, STL); if (vm < 80) continue;
    const step = 110, dc = PAL.c.dust;
    h.members.forEach((m, mi) => {
      for (let k = Math.floor(Dm / step); k > Math.floor(Dm / step) - 9; k--) {
        const age = (Dm - k * step) / vm; if (age < 0 || age > 1.6) continue;
        const f = h.fx0 + h.dir * (k * step) - SD + (m.dfx || 0) - h.dir * 40, z = herdZ(h, m, mi, STL) + (hh(k, mi) - 0.5) * 60;
        if (tooClose(f, z)) continue;
        const p = P3w(f, z, 20 + 70 * age), q = P3w(f, z, 120 + 70 * age), kk = Math.abs(p[1] - q[1]) / (100 * sc + 1e-6);
        const r = (40 + 110 * age) * kk * sc * (h.sp.L > 100 ? 1.4 : 0.9), al = 0.22 * h.dust * (1 - age / 1.6) * clamp(vm / 400);
        const g = ctx.createRadialGradient(p[0], p[1], 0, p[0], p[1], r); g.addColorStop(0, col(dc, al)); g.addColorStop(1, col(dc, 0));
        ctx.fillStyle = g; ctx.fillRect(p[0] - r, p[1] - r, 2 * r, 2 * r);
      }
    });
  }
}
// weather of the land: fireflies, snow, falling petals or leaves, rain
function drawWeather(t) {
  ctx.setTransform(1, 0, 0, 1, 0, 0);
  if (PAL.ff > 0.02) { ctx.globalAlpha = PAL.ff; drawFireflies(t); ctx.globalAlpha = 1; }
  if (PAL.snow > 0.02) {
    ctx.fillStyle = `rgba(255,255,255,${0.85 * PAL.snow})`;
    for (let i = 0; i < 170; i++) {
      const r = (1.6 + 3.6 * hh(i, 4.1)) * sc, x = ((hh(i, 1.1) * W + Math.sin(t * 0.7 + i) * 30 * sc + t * 25 * sc) % W + W) % W, y = (hh(i, 2.3) * H + t * (70 + 110 * hh(i, 3.2)) * sc) % H;
      ctx.beginPath(); ctx.arc(x, y, r, 0, TAU); ctx.fill();
    }
  }
  if (PAL.petals > 0.02) {
    ctx.fillStyle = col(PAL.c.petalCol, 0.85 * PAL.petals);
    for (let i = 0; i < 60; i++) {
      const x = ((hh(i, 5.1) * W + t * (30 + 50 * hh(i, 6.2)) * sc + Math.sin(t * 1.3 + i) * 40 * sc) % W + W) % W, y = (hh(i, 7.7) * H + t * (60 + 70 * hh(i, 8.3)) * sc) % H;
      ctx.beginPath(); ctx.ellipse(x, y, 8 * sc, 3.6 * sc, t * (1 + 2 * hh(i, 9.4)) + i, 0, TAU); ctx.fill();
    }
  }
  if (PAL.rain > 0.02) {
    ctx.strokeStyle = `rgba(225,238,255,${0.32 * PAL.rain})`; ctx.lineWidth = 1.6 * sc;
    ctx.beginPath();
    for (let i = 0; i < 130; i++) {
      const x = ((hh(i, 3.3) * W + t * 90 * sc) % W + W) % W, y = (hh(i, 4.4) * H + t * 1500 * sc * (0.8 + 0.4 * hh(i, 5.5))) % H;
      ctx.moveTo(x, y); ctx.lineTo(x - 6 * sc, y + 38 * sc);
    }
    ctx.stroke();
  }
}

const GRAIN = (() => { const c = document.createElement('canvas'); c.width = c.height = 256; const g = c.getContext('2d'), im = g.createImageData(256, 256), r = rng(9);
  for (let i = 0; i < 256 * 256; i++) { const v = 128 + (r() - 0.5) * 255; im.data[i * 4] = im.data[i * 4 + 1] = im.data[i * 4 + 2] = v; im.data[i * 4 + 3] = 255; } g.putImageData(im, 0, 0); return c; })();
function drawCine() {
  ctx.setTransform(1, 0, 0, 1, 0, 0); ctx.globalAlpha = 1;
  const hz = CFG.haze ? keyVal(CFG.haze, STL) : 0;
  if (hz > 0.01) { const g = ctx.createLinearGradient(0, 0, 0, H); g.addColorStop(0, col(PAL.c.dust, 0.15 * hz)); g.addColorStop(0.6, col(PAL.c.dust, 0.45 * hz)); g.addColorStop(1, col(PAL.c.dust, 0.3 * hz)); ctx.fillStyle = g; ctx.fillRect(0, 0, W, H); }
  if (CFG.grade) { ctx.globalCompositeOperation = 'soft-light'; ctx.globalAlpha = CFG.grade[1]; ctx.fillStyle = CFG.grade[0]; ctx.fillRect(0, 0, W, H); ctx.globalAlpha = 1; }
  if (CFG.glow) { ctx.globalCompositeOperation = 'screen'; const g = ctx.createRadialGradient(W * CFG.glow[0], H * 0.05, 0, W * CFG.glow[0], H * 0.05, H * 0.7); g.addColorStop(0, `rgba(${CFG.glow[1]},0.28)`); g.addColorStop(1, `rgba(${CFG.glow[1]},0)`); ctx.fillStyle = g; ctx.fillRect(0, 0, W, H); }
  ctx.globalCompositeOperation = 'source-over';
  vignette();
  ctx.globalCompositeOperation = 'overlay'; ctx.globalAlpha = 0.09;
  const gx = (hh(Math.floor(STL * 30), 1.3) * 256) | 0, gy = (hh(Math.floor(STL * 30), 2.7) * 256) | 0;
  for (let y = -gy; y < H; y += 256) for (let x = -gx; x < W; x += 256) ctx.drawImage(GRAIN, x, y);
  ctx.globalCompositeOperation = 'source-over'; ctx.globalAlpha = 1;
  const bh = Math.round(H * 0.035); ctx.fillStyle = '#000'; ctx.fillRect(0, 0, W, bh); ctx.fillRect(0, H - bh, W, bh);
}
