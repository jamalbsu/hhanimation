// the Starcompass: a glowing feather needle in a ring, bottom right
function drawCompass() {
  const C = CFG.compass; if (!C) return;
  const a = smooth(clamp((STL - 2.6) / 0.5)) * (1 - smooth(clamp((STL - ENDAT) / 0.3))); if (a <= 0) return;
  const r = S * 0.075, x = W - S * 0.06 - r, y = H - H * 0.035 - S * 0.07 - r, col = colAt(C.col || [[0, '#ffd76a']], STL).map(v => v | 0).join(',');
  let ang = keyVal(C.ang, STL) * DEG; for (const [t0, t1] of C.spin || []) if (STL > t0) ang += smooth(clamp((STL - t0) / (t1 - t0))) * TAU * 3;
  ctx.save(); ctx.globalAlpha = a; ctx.translate(x, y);
  ctx.globalCompositeOperation = 'lighter'; hzPuff(0, 0, r * 1.7, `rgba(${col},0.35)`, `rgba(${col},0.12)`, `rgba(${col},0)`); ctx.globalCompositeOperation = 'source-over';
  ctx.fillStyle = 'rgba(10,8,22,0.72)'; ctx.beginPath(); ctx.arc(0, 0, r, 0, TAU); ctx.fill();
  ctx.strokeStyle = `rgba(${col},0.95)`; ctx.lineWidth = S * 0.006; ctx.stroke();
  ctx.lineWidth = S * 0.003; for (let k = 0; k < 8; k++) { const q = k * TAU / 8, l = k % 2 ? 0.1 : 0.18; ctx.beginPath(); ctx.moveTo(Math.cos(q) * r * (1 - l), Math.sin(q) * r * (1 - l)); ctx.lineTo(Math.cos(q) * r * 0.97, Math.sin(q) * r * 0.97); ctx.stroke(); }
  ctx.rotate(ang);                                                      // the feather needle
  const g = ctx.createLinearGradient(0, r * 0.8, 0, -r * 0.85); g.addColorStop(0, 'rgba(255,255,255,0.4)'); g.addColorStop(1, `rgba(${col},1)`);
  ctx.fillStyle = g; ctx.beginPath(); ctx.moveTo(0, -r * 0.85); ctx.quadraticCurveTo(r * 0.24, -r * 0.2, 0, r * 0.62); ctx.quadraticCurveTo(-r * 0.24, -r * 0.2, 0, -r * 0.85); ctx.fill();
  ctx.strokeStyle = 'rgba(255,255,255,0.8)'; ctx.lineWidth = S * 0.0025; ctx.beginPath(); ctx.moveTo(0, -r * 0.8); ctx.lineTo(0, r * 0.7); ctx.stroke();
  for (let k = 1; k < 6; k++) { const yy = -r * 0.7 + k * r * 0.22; ctx.beginPath(); ctx.moveTo(0, yy); ctx.lineTo(r * 0.16, yy - r * 0.1); ctx.moveTo(0, yy); ctx.lineTo(-r * 0.16, yy - r * 0.1); ctx.stroke(); }
  ctx.rotate(-ang); ctx.fillStyle = '#ffffff'; ctx.beginPath(); ctx.arc(0, 0, S * 0.006, 0, TAU); ctx.fill();
  ctx.fillStyle = `rgba(${col},0.9)`; ctx.textAlign = 'center'; ctx.font = `700 ${Math.round(S * 0.016)}px "Segoe UI", system-ui, sans-serif`; try { ctx.letterSpacing = `${Math.round(S * 0.005)}px`; } catch (e) {}
  ctx.fillText('STARCOMPASS', 0, r + S * 0.03); try { ctx.letterSpacing = '0px'; } catch (e) {}
  ctx.restore();
}

const CRIT = (CFG.critters || []).map((c, gi) => { const r = rng(100 + gi), a = []; for (let i = 0; i < (c.n || 1); i++) a.push({ df: lerp(c.f[0], c.f[1], r()), z: lerp(c.z[0], c.z[1], r()), ph: r() * TAU, s: 0.8 + r() * 0.4 }); return { ...c, D: worldD(c.at || 0, 0), m: a }; });
function critPos(c, m, i) {
  const age = STL - (c.t0 === undefined ? -99 : c.t0);
  let f = c.D + m.df - SD, z = m.z, hg = c.hg || 0;
  if (c.kind === 'crab') z += 60 * Math.sin(STL * 2 + m.ph);
  if (c.kind === 'turtle' && age > 0) z -= age * (c.vz || 90);
  if (c.kind === 'tumbleweed') { z += ((STL * (c.vz || 600) + m.ph * 400) % 5200) - 2600; hg = Math.abs(Math.sin(STL * 3 + m.ph)) * 60; }
  if (c.kind === 'gull') { const a = STL * 0.9 + m.ph; f += 300 * Math.cos(a); z += 300 * Math.sin(a); hg = 380 + 120 * Math.sin(STL * 1.3 + m.ph); if (c.dive && STL > c.dive[0] && STL < c.dive[1]) hg -= 220 * Math.sin(Math.PI * (STL - c.dive[0]) / (c.dive[1] - c.dive[0])); }
  return { f, z, hg };
}
function drawCritters(front) {
  CRIT.forEach(c => c.m.forEach((m, i) => {
    if (c.until !== undefined && STL > c.until) return; if (c.from !== undefined && STL < c.from) return;
    const p = critPos(c, m, i); if (!hzVis(p.f, p.z) || (hzDepth(p.f, p.z) < -40) !== front) return;
    const q = P3s(p.f, p.hg, p.z), s = (c.size || 22) * 2.3 * m.s * hzK(p.f, p.hg, p.z); if (s < 0.7) return;
    SPR[c.kind](q[0], q[1], s, STL, i);
  }));
}
// riders: a small friend on a cat's back
function drawRiders() {
  for (const r of CFG.riders || []) {
    if (STL < r.from || (r.until !== undefined && STL > r.until)) continue;
    const a = r.cat === undefined ? { fx: 0, z: 0, size: 1 } : OURS[r.cat], f = a.fx + (r.dx || -10), z = a.z, hg = 205 * (a.size || 1) + 6 * Math.abs(Math.sin(STL * 9));
    if (!hzVis(f, z)) continue; const q = P3s(f, hg, z), s = (r.size || 16) * 2.2 * hzK(f, hg, z);
    SPR[r.kind](q[0], q[1], s, STL, r.i || 0);
  }
}

// dust devils: spinning funnels of sand
function twisterAt(T) { return { f: keyVal(T.fx, STL), z: keyVal(T.z, STL) }; }
function drawTwister(T, t) {
  const { f, z } = twisterAt(T); if (!hzVis(f, z)) return; const H0 = T.h || 1500;
  ctx.globalCompositeOperation = 'source-over';
  for (let k = 0; k <= 22; k++) {
    const u = k / 22, hg = u * H0, r = (50 + 520 * Math.pow(u, 1.6)) * (1 + 0.08 * Math.sin(t * 6 + k)), sway = 90 * Math.sin(t * 1.4 + u * 3) * u, p = P3s(f + sway, hg, z), kk = hzK(f, hg, z);
    const rx = r * kk, ry = rx * 0.28; if (rx < 1) continue;
    ctx.globalAlpha = clamp(1.5 - rx / (W * 0.45), 0.22, 1);             // right on top of the camera it thins out, so the pack stays readable
    const g = ctx.createRadialGradient(p[0] - rx * 0.25, p[1], rx * 0.05, p[0], p[1], rx); g.addColorStop(0, `rgba(214,176,124,${0.55 + 0.2 * (1 - u)})`); g.addColorStop(0.5, `rgba(150,106,64,${0.62 + 0.25 * (1 - u)})`); g.addColorStop(0.85, 'rgba(118,82,50,0.4)'); g.addColorStop(1, 'rgba(118,82,50,0)');
    ctx.fillStyle = g; ctx.beginPath(); ctx.ellipse(p[0], p[1], rx, ry + rx * 0.2, 0, 0, TAU); ctx.fill();
    ctx.strokeStyle = `rgba(90,60,36,${0.5 * (1 - u * 0.5)})`; ctx.lineWidth = Math.max(1, 3.5 * kk); ctx.beginPath(); ctx.ellipse(p[0], p[1], rx * 0.8, ry, 0, t * 5 + k, t * 5 + k + 2.2); ctx.stroke();
    ctx.strokeStyle = `rgba(255,232,190,${0.3 * (1 - u * 0.4)})`; ctx.lineWidth = Math.max(1, 2 * kk); ctx.beginPath(); ctx.ellipse(p[0], p[1], rx * 0.7, ry * 0.9, 0, t * 5 + k + 3.4, t * 5 + k + 4.6); ctx.stroke();
  }
  ctx.globalAlpha = 1; ctx.fillStyle = 'rgba(84,58,36,0.9)';              // twigs and grit whirling around it

}

// a gorge across the road, bridged by planks that snap after the pack is over
const GORGE = CFG.gorge ? { ...CFG.gorge, D: worldD(CFG.gorge.at, 0) } : null;
function drawGorge() {
  if (!GORGE) return; const fa = GORGE.D - SD, fb = fa + GORGE.w, ga = clamp((5200 - fa) / 2200); if (ga <= 0) return;   // it fades in out of the haze
  ctx.save(); ctx.globalAlpha = ga;
  hzFill(hzBand(fa - 50, fb + 50, -6000, 6000, 30), '#6a4a2c');                    // the sunlit lip
  hzFill(hzBand(fa - 10, fb + 10, -6000, 6000, 30), '#3a2818');                    // rock walls going down
  hzFill(hzBand(fa + GORGE.w * 0.2, fb - GORGE.w * 0.2, -6000, 6000, 30), '#1a1410');  // the shadowed depths
  ctx.restore();
  const snap = GORGE.snap !== undefined ? STL - GORGE.snap : -1;
  for (let k = 0; k < 12; k++) {
    const u0 = k / 12, f0 = lerp(fa - 30, fb + 30, u0), f1 = f0 + (fb - fa + 60) / 12 - 8, drop = snap > 0 ? 900 * snap * snap * (0.6 + 0.8 * hh(k, 4)) : 0; if (drop > 2500) continue;

  }
  ctx.globalAlpha = ga; ctx.strokeStyle = '#d8c090'; ctx.lineWidth = Math.max(1, 3 * sc);  for (const z of [-250, 250]) { ctx.beginPath(); let first = true; for (let u = 0; u <= 1.001; u += 0.1) { const f = lerp(fa - 60, fb + 60, u), hg = (snap > 0 && u > 0.1 && u < 0.9 ? -900 * snap * snap : 0) + 90 - 40 * Math.sin(Math.PI * u); if (!hzVis(f, z)) { first = true; continue; } const p = P3s(f, hg, z); first ? ctx.moveTo(p[0], p[1]) : ctx.lineTo(p[0], p[1]); first = false; } ctx.stroke(); }
  ctx.globalAlpha = 1;
}
