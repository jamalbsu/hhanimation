// ---- DISASTERS: wildfire, flash flood, a crocodile ford, a rockslide and a lightning storm ----
const hzK = (f, hg, z) => sc * camK0(f, -hg, z) / camK0(0, PIV, 0);          // screen px per body unit at a world point
const hzDepth = (f, z) => (f * RS - z * RC) * CT;
const hzVis = (f, z) => CH_D + hzDepth(f, z) > CH_D * 0.32;
function hzClip(fa, fb, z) {                                                  // the part of [fa, fb] on lane z in front of the camera
  const lim = -0.62 * CH_D, da = hzDepth(fa, z), db = hzDepth(fb, z);
  if (da > lim && db > lim) return [fa, fb];
  if (da <= lim && db <= lim) return null;
  const fc = fa + (fb - fa) * (lim - da) / (db - da);
  return da > lim ? [fa, fc] : [fc, fb];
}
function hzBand(fa, fb, za, zb, n, hg) {                                      // a flat patch of ground, as a screen polygon
  const A = [], B = [];
  for (let i = 0; i <= n; i++) { const z = lerp(za, zb, i / n), c = hzClip(fa, fb, z); if (!c) continue; A.push(P3s(c[0], hg || 0, z)); B.push(P3s(c[1], hg || 0, z)); }
  return A.length > 1 ? A.concat(B.reverse()) : null;
}
function hzFill(P, style) { if (!P) return; ctx.fillStyle = style; ctx.beginPath(); ctx.moveTo(P[0][0], P[0][1]); for (let i = 1; i < P.length; i++) ctx.lineTo(P[i][0], P[i][1]); ctx.closePath(); ctx.fill(); }
function hzRing(f, z, rf, rz, hg) { const P = []; for (let i = 0; i < 14; i++) { const a = TAU * i / 14; P.push(P3s(f + Math.cos(a) * rf, hg || 0, z + Math.sin(a) * rz)); } return P; }
function hzPuff(x, y, r, c0, c1, a) { const g = ctx.createRadialGradient(x - r * 0.2, y - r * 0.25, r * 0.05, x, y, r); g.addColorStop(0, c0); g.addColorStop(0.6, c1); g.addColorStop(1, a); ctx.fillStyle = g; ctx.fillRect(x - r, y - r, 2 * r, 2 * r); }
const WT = CFG.wall ? (CFG.wall.type || 'snow') : null;

// ---- the river: a ford across the road, with muddy banks ----
const RIV = CFG.river ? { d0: tab(TAB0.d, CFG.river.at), w: CFG.river.w, col: CFG.river.col || ['#3d7f8c', '#2a5866'], bank: CFG.river.bank || '#7a6448' } : null;
const rivF = () => RIV.d0 - SD;
const inRiver = f => !!RIV && f > rivF() - 30 && f < rivF() + RIV.w + 30;
function hzWallF() { let f = keyVal(CFG.wall.fx, STL); if (RIV && WT === 'fire') f = Math.min(f, RIV.d0 - 160 - SD); return f; }
function HZ_NOTREE(f, z) {                                                    // no trees standing in water, in the riverbed or in the burnt land
  if (RIV && f > rivF() - 220 && f < rivF() + RIV.w + 220) return true;
  if (CFG.bed && Math.abs(z) < CFG.bed.w + 200) return true;
  if (WT === 'water' && f < hzWallF() + 100) return true;
  if (WT === 'fire' && f < hzWallF() - 40) return true;
  return false;
}
function drawRiver(t) {
  if (!RIV) return; const fa = rivF(), fb = fa + RIV.w; if (fb < -12000 || fa > 12000) return;
  hzFill(hzBand(fa - 110, fb + 110, -6000, 6000, 40), RIV.bank);
  hzFill(hzBand(fa, fb, -6000, 6000, 40), RIV.col[0]);
  if (RIV.w > 600) hzFill(hzBand(fa + 220, fb - 220, -6000, 6000, 40), RIV.col[1]);
  ctx.strokeStyle = 'rgba(255,255,255,0.32)'; ctx.lineWidth = Math.max(1, 2.2 * sc); ctx.beginPath();
  const n = Math.min(260, 40 + RIV.w / 40);
  for (let i = 0; i < n; i++) {
    const f = fa + hh(i, 1.3) * RIV.w, l = 120 + 220 * hh(i, 4.2), z = ((hh(i, 2.7) * 9000 + t * 300 * (0.6 + 0.8 * hh(i, 3.1))) % 9000) - 4500;
    if (!hzVis(f, z) || !hzVis(f, z + l)) continue; const a = P3s(f, 0, z), b = P3s(f, 0, z + l); ctx.moveTo(a[0], a[1]); ctx.lineTo(b[0], b[1]);
  }
  ctx.stroke();
}
function hzWaders() {                                                         // everyone standing in the water: where and how big
  const pts = [{ f: 0, z: 0, k: 1 }]; OURS.forEach(a => pts.push({ f: a.fx, z: a.z, k: a.size }));
  for (const h of HERDS) h.members.forEach((m, mi) => pts.push({ f: herdFx(h, m, STL), z: herdZ(h, m, mi, STL), k: (m.size || 1) * h.sp.L / 85 }));
  return pts;
}
function drawSplashes(t) {
  const water = RIV ? RIV.col[0] : null, flooded = WT === 'water' ? hzWallF() : -1e9;
  for (const p of hzWaders()) {
    const wet = inRiver(p.f) || p.f < flooded; if (!wet || !hzVis(p.f, p.z)) continue;
    ctx.fillStyle = water ? col(rgbOf(water), 0.75) : 'rgba(110,84,52,0.8)'; fillPoly(ctx, hzRing(p.f, p.z, 95 * p.k, 55 * p.k));
    ctx.strokeStyle = 'rgba(255,255,255,0.75)'; ctx.lineWidth = Math.max(1, 2.4 * sc); ctx.stroke();
    ctx.fillStyle = 'rgba(240,250,255,0.85)';
    for (let i = 0; i < 10; i++) {
      const ph = frac(t * 2.3 + hh(i, (p.f * 0.013 + p.z * 0.017) % 97)), a = TAU * hh(i, 5.5), r = (40 + 90 * ph) * p.k;
      const q = P3s(p.f + Math.cos(a) * r, 150 * p.k * Math.sin(Math.PI * ph), p.z + Math.sin(a) * r * 0.6), rr = Math.max(1, 6.5 * (1 - ph) * hzK(p.f, 0, p.z) * p.k);
      ctx.beginPath(); ctx.arc(q[0], q[1], rr, 0, TAU); ctx.fill();
    }
  }
}

// ---- crocodiles: low shapes in the water that rear up and snap ----
const CROCS = (CFG.crocs || []).map(c => ({ ...c, size: c.size || 1, d: c.d !== undefined ? c.d : tab(TAB0.d, c.at) - (RIV ? RIV.d0 : 0) + (c.df || 0) }));
function crocState(c) {
  const f = (RIV ? RIV.d0 : 0) + c.d - SD, side = c.z > 0 ? 1 : -1, l = c.lunge ? STL - c.at : -9;
  const lk = l > -0.25 && l < 1.2 ? (l < 0 ? 0 : l < 0.22 ? smooth(l / 0.22) : l < 0.75 ? 1 : 1 - smooth((l - 0.75) / 0.45)) : 0;
  const drift = 30 * Math.sin(STL * 0.7 + c.d * 0.01);
  return { f: f + drift, z: c.z - side * (c.reach || 230) * lk, side, lk, l, open: l > -0.05 && l < 0.8 ? Math.sin(Math.PI * clamp((l + 0.05) / 0.62)) : 0 };
}
function drawCroc(c, s) {
  const L = 140 * c.size, dz = -s.side, at = (u, v, hg) => P3s(s.f + v, hg, s.z + dz * u);
  const lift = u => s.lk * 80 * c.size * clamp((u + 0.25 * L) / (0.85 * L)), hk = hzK(s.f, 0, s.z) * c.size;
  const prof = [[0.66, 0.045], [0.5, 0.1], [0.32, 0.13], [0.16, 0.2], [-0.1, 0.23], [-0.32, 0.17], [-0.56, 0.09], [-0.85, 0.035], [-1.0, 0]];
  const body = [];
  for (const [u, w] of prof) body.push(at(u * L, w * L, lift(u * L)));
  for (let i = prof.length - 1; i >= 0; i--) body.push(at(prof[i][0] * L, -prof[i][1] * L, lift(prof[i][0] * L)));
  ctx.globalAlpha = 0.5 + 0.5 * s.lk; ctx.fillStyle = '#46552e'; fillPoly(ctx, body); ctx.globalAlpha = 1;
  ctx.fillStyle = '#28331a';
  for (let k = 0; k < 11; k++) { const u = lerp(0.18, -0.9, k / 10) * L, p = at(u, 0, lift(u) + 6); ctx.beginPath(); ctx.arc(p[0], p[1], Math.max(1, 5.5 * hk), 0, TAU); ctx.fill(); }
  for (const v of [-1, 1]) {                                                   // the eyes that watch from the water
    const e = at(0.34 * L, v * 0.075 * L, lift(0.34 * L) + 12);
    ctx.fillStyle = '#46552e'; ctx.beginPath(); ctx.arc(e[0], e[1], 8 * hk, 0, TAU); ctx.fill();
    ctx.fillStyle = '#f2d84a'; ctx.beginPath(); ctx.arc(e[0], e[1] - hk, 3.6 * hk, 0, TAU); ctx.fill(); ctx.fillStyle = '#111'; ctx.fillRect(e[0] - 0.8 * hk, e[1] - 4 * hk, 1.6 * hk, 6 * hk);
  }
  if (s.open > 0.02) {                                                         // the jaws, drawn flat on screen so they read from any side
    const B0 = at(0.28 * L, 0, lift(0.28 * L) + 10), T = at(0.7 * L, 0, lift(0.7 * L) + 10), dx = T[0] - B0[0], dy = T[1] - B0[1], ln = Math.hypot(dx, dy);
    if (ln > 6) {
      let nx = -dy / ln, ny = dx / ln; if (ny > 0) { nx = -nx; ny = -ny; }
      const g = s.open * Math.max(ln * 0.55, 30 * hk), w = Math.max(5, 0.07 * L * hk);
      const UP = [[B0[0] - nx * w, B0[1] - ny * w], [B0[0] + nx * w, B0[1] + ny * w], [T[0] + nx * g, T[1] + ny * g]];
      const LO = [[B0[0] - nx * w, B0[1] - ny * w], [B0[0] + nx * w * 0.3, B0[1] + ny * w * 0.3], [T[0] - nx * g * 0.45, T[1] - ny * g * 0.45]];
      ctx.fillStyle = '#c84a52'; fillPoly(ctx, [B0, [T[0] + nx * g, T[1] + ny * g], [T[0] - nx * g * 0.45, T[1] - ny * g * 0.45]]);
      ctx.fillStyle = '#fff6e0';
      for (const [J, sg] of [[UP, -1], [LO, 1]]) for (let k = 1; k < 7; k++) {
        const q = k / 7, a = [lerp(B0[0], J[2][0], q), lerp(B0[1], J[2][1], q)], tw = Math.max(2, ln * 0.035);
        ctx.beginPath(); ctx.moveTo(a[0] - dx / ln * tw, a[1] - dy / ln * tw); ctx.lineTo(a[0] + dx / ln * tw, a[1] + dy / ln * tw); ctx.lineTo(a[0] + sg * nx * tw * 2.2, a[1] + sg * ny * tw * 2.2); ctx.closePath(); ctx.fill();
      }
      ctx.fillStyle = '#4e5e32'; fillPoly(ctx, UP); ctx.fillStyle = '#9a9a68'; fillPoly(ctx, LO);
    }
    ctx.fillStyle = 'rgba(240,250,255,0.9)';                                    // the strike throws water
    for (let i = 0; i < 18; i++) { const ph = clamp(s.l / 0.9), a = TAU * hh(i, 7.1), r = (60 + 220 * ph) * c.size, q = P3s(s.f + Math.cos(a) * r, 260 * c.size * Math.sin(Math.PI * ph) * hh(i, 8.2), s.z + Math.sin(a) * r * 0.6); ctx.beginPath(); ctx.arc(q[0], q[1], Math.max(1, 8 * (1 - ph) * hk), 0, TAU); ctx.fill(); }
  }
}
function crocAnchor(i) { const c = CROCS[i]; if (!c) return null; const s = crocState(c); return hzVis(s.f, s.z) ? P3s(s.f, 80, s.z) : null; }

// ---- the rockslide: a cliff along the road that drops bouncing boulders ----
const CLIFF = CFG.cliff ? { z: CFG.cliff.z, h: CFG.cliff.h, end: CFG.cliff.until ? tab(TAB0.d, CFG.cliff.until) : 1e12 } : null;
const ROCKS = (CFG.rocks || []).map(([tc, off, r], i) => ({ tc, D: tab(TAB0.d, tc) + off, r, v: CFG.rockV || 950, seed: i * 7.31 + 1.7 }));
function rockTimes(k) { const z0 = CLIFF ? CLIFF.z + 50 : -950, zL = z0 + 300, tL = k.tc - (0 - zL) / k.v; return { z0, zL, tL, tS: tL - 0.85 }; }
function rockState(k) {
  const { z0, zL, tL, tS } = rockTimes(k), h0 = CLIFF ? CLIFF.h + 60 : 900, t = STL; if (t < tS || t > k.tc + 3.6) return null;
  let z, hg;
  if (t < tL) { const q = (t - tS) / 0.85; z = lerp(z0, zL, q); hg = h0 * (1 - q * q); }
  else { z = zL + (t - tL) * k.v; const ph = (z - zL) / 320; hg = Math.abs(Math.sin(Math.PI * ph)) * 170 * Math.exp(-ph * 0.32) * (k.r / 90); }
  return { f: k.D - SD, z, hg, rot: z / k.r + k.seed };
}
function drawRock(k, s) {
  const c = P3s(s.f, s.hg + k.r, s.z), rad = k.r * hzK(s.f, s.hg + k.r, s.z); if (rad < 0.5) return;
  const sh = hzRing(s.f, s.z, k.r * 1.1, k.r * 0.9);
  ctx.fillStyle = `rgba(30,18,8,${0.32 * clamp(1 - s.hg / 700)})`; fillPoly(ctx, sh);
  if (s.hg < 30) hzPuff(c[0], c[1] + rad * 0.6, rad * 2.4, 'rgba(214,184,150,0.5)', 'rgba(214,184,150,0.25)', 'rgba(214,184,150,0)');
  const P = []; for (let j = 0; j < 10; j++) { const a = s.rot + TAU * j / 10, rr = rad * (0.78 + 0.28 * hh(j, k.seed)); P.push([c[0] + Math.cos(a) * rr, c[1] + Math.sin(a) * rr]); }
  const g = ctx.createRadialGradient(c[0] - rad * 0.4, c[1] - rad * 0.45, rad * 0.1, c[0], c[1], rad * 1.1); g.addColorStop(0, '#c8ab8c'); g.addColorStop(0.55, '#8a6a52'); g.addColorStop(1, '#4a3628');
  ctx.fillStyle = g; fillPoly(ctx, P); ctx.strokeStyle = '#2e2018'; ctx.lineWidth = Math.max(1, rad * 0.05); ctx.stroke();
  ctx.strokeStyle = 'rgba(60,40,28,0.6)'; ctx.beginPath(); ctx.moveTo(P[1][0] * 0.6 + c[0] * 0.4, P[1][1] * 0.6 + c[1] * 0.4); ctx.lineTo(c[0], c[1]); ctx.lineTo(P[6][0] * 0.7 + c[0] * 0.3, P[6][1] * 0.7 + c[1] * 0.3); ctx.stroke();
}
function rockAnchor(i) { const k = ROCKS[i]; if (!k) return null; const s = rockState(k); return s && hzVis(s.f, s.z) ? P3s(s.f, s.hg + k.r * 2.3, s.z) : null; }
function drawCliff() {
  if (!CLIFF) return; const z = CLIFF.z, step = 240, k0 = Math.floor((SD - 7000) / step);
  for (let k = k0; k < k0 + 80; k++) {
    const fa = k * step - SD, fb = fa + step; if (fa + SD > CLIFF.end) break;
    if (!hzVis(fa, z) || !hzVis(fb, z)) continue;
    const end = fb + SD > CLIFF.end, ha = CLIFF.h * (0.82 + 0.3 * hh(k, 1.7)), hb = end ? 0 : CLIFF.h * (0.82 + 0.3 * hh(k + 1, 1.7));
    ctx.fillStyle = k % 2 ? '#94603e' : '#835436'; fillPoly(ctx, [P3s(fa, 0, z), P3s(fa, ha, z), P3s(fb, hb, z), P3s(fb, 0, z)]);
    for (const [q, cl] of [[0.18, 'rgba(60,34,20,0.5)'], [0.36, 'rgba(196,140,100,0.45)'], [0.55, 'rgba(60,34,20,0.45)'], [0.72, 'rgba(196,150,110,0.4)'], [0.9, 'rgba(50,28,16,0.6)']]) {
      ctx.fillStyle = cl; fillPoly(ctx, [P3s(fa, ha * q, z), P3s(fa, ha * (q + 0.05), z), P3s(fb, hb * (q + 0.05), z), P3s(fb, hb * q, z)]);
    }
  }
}

  for (let i = 0; i < 70; i++) { const f = fx0 - 150 - hh(i, 2.2) * 2600, z = (hh(i, 3.3) - 0.5) * 6000; if (!hzVis(f, z)) continue; const p = P3s(f, 0, z), r = (10 + 20 * hh(i, 4.4)) * hzK(f, 0, z) * (0.6 + 0.4 * Math.sin(t * 5 + i)); ctx.beginPath(); ctx.arc(p[0], p[1], Math.max(0.5, r), 0, TAU); ctx.fill(); }
  const items = [];
  for (let i = 0; i < 28; i++) for (let l = 0; l < 3; l++) {

  for (let l = 0; l < 2; l++) for (let i = 0; i < 58; i++) {

function drawFlood(t) {
  const fx0 = hzWallF();

// ---- the storm: darkness, heavy rain and lightning ----
const BOLTS = CFG.bolts || [];
function boltFlash() { let fl = 0; for (const b of BOLTS) { const d = STL - b.t; if (d >= 0 && d < 0.7) fl = Math.max(fl, (b.big ? 1 : 0.6) * (d < 0.06 ? 1 : d < 0.12 ? 0.3 : d < 0.2 ? 0.85 : Math.exp(-(d - 0.2) * 6))); } return fl; }
function drawBolt(b, d) {
  const r = rng(Math.floor(b.t * 1000) + 3), yEnd = H * (b.y || 0.5), seg = 16, main = [[W * b.x, 0]];
  for (let i = 1; i <= seg; i++) main.push([main[i - 1][0] + (r() - 0.5) * S * 0.09, yEnd * i / seg]);
  const paths = [main];
  for (let k = 0; k < 4; k++) { const i0 = 2 + Math.floor(r() * (seg - 5)), br = [main[i0]], dir = r() < 0.5 ? -1 : 1; for (let j = 1; j < 6; j++) br.push([br[j - 1][0] + dir * r() * S * 0.05, br[j - 1][1] + yEnd / seg * (0.6 + r() * 0.5)]); paths.push(br); }
  const a = 1 - d / 0.3;
  ctx.save(); ctx.lineJoin = 'round'; ctx.lineCap = 'round'; ctx.shadowColor = 'rgba(170,190,255,1)'; ctx.shadowBlur = S * 0.05;
  paths.forEach((p, k) => { ctx.strokeStyle = `rgba(235,240,255,${a})`; ctx.lineWidth = S * (k ? 0.003 : 0.008) * (b.big ? 1.3 : 1); ctx.beginPath(); ctx.moveTo(p[0][0], p[0][1]); for (const q of p) ctx.lineTo(q[0], q[1]); ctx.stroke(); });
  ctx.restore();
}
function drawStorm(t) {
  ctx.setTransform(1, 0, 0, 1, 0, 0);
  const fl = boltFlash(), dk = kv(CFG.dark, 0), rn = kv(CFG.rain, 0);
  if (dk > 0.01) { ctx.fillStyle = `rgba(6,10,26,${dk * (1 - 0.9 * fl)})`; ctx.fillRect(0, 0, W, H); }
  if (rn > 0.01) {
    ctx.strokeStyle = `rgba(210,225,255,${0.38 * rn})`; ctx.lineWidth = Math.max(1, 1.8 * sc); ctx.beginPath();
    for (let i = 0; i < 260; i++) { const x = ((hh(i, 3.3) * W * 1.2 + t * 140 * sc) % (W * 1.2)) - W * 0.1, y = (hh(i, 4.4) * H + t * 1900 * sc * (0.8 + 0.4 * hh(i, 5.5))) % H, l = 46 * sc * (0.7 + 0.6 * hh(i, 6.6)); ctx.moveTo(x, y); ctx.lineTo(x - l * 0.22, y + l); }
    ctx.stroke();
  }
  if (fl > 0.01) { ctx.globalCompositeOperation = 'screen'; ctx.fillStyle = `rgba(185,200,255,${0.6 * fl})`; ctx.fillRect(0, 0, W, H); ctx.globalCompositeOperation = 'source-over'; }
  for (const b of BOLTS) { const d = STL - b.t; if (d < 0 || d > 0.3 || (d > 0.07 && d < 0.12)) continue; drawBolt(b, d); }
}
