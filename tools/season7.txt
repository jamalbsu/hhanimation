// actors: any sprite on a keyed path through the world. at = the moment the road position is fixed; df / z / hg are [[t, value]] keys
const ACT = (CFG.actors || []).map(a => ({ ...a, D: worldD(a.at || 0, 0) }));
function drawActors(front) {
  for (const a of ACT) {
    if ((a.from !== undefined && STL < a.from) || (a.until !== undefined && STL > a.until)) continue;
    const f = a.D + keyVal(a.df, STL) - SD, z = keyVal(a.z, STL), hg = a.hg ? keyVal(a.hg, STL) : 0;
    if (!hzVis(f, z) || (hzDepth(f, z) < -40) !== front) continue;
    const q = P3s(f, hg, z), s = (a.size || 22) * 2.3 * hzK(f, hg, z); if (s < 0.7) continue;
    SPR[a.kind](q[0], q[1], s, STL, a.i || 0);
  }
}
// the cliff edge: from "from" on, the ground ahead of "at" simply ends and a green valley lies far below
function drawEdge() {
  const E = CFG.edge; if (!E || STL < E.from) return; const k = smooth(clamp((STL - E.from) / 0.7)), f0 = worldD(E.at, 0) - SD;
  ctx.globalAlpha = k; hzFill(hzBand(f0, f0 + 30000, -9000, 9000, 24), E.col || '#a8c8cc'); hzFill(hzBand(f0 + 400, f0 + 30000, -9000, 9000, 24), E.col2 || '#8fb8bc');
  hzFill(hzBand(f0 - 320, f0, -9000, 9000, 6), '#3a2c24'); ctx.globalAlpha = 1;
}
// a flash flood coming down the riverbed at the pack: a brown front, foam and torn reeds
function drawFlash(t) {
  const F = CFG.flood; if (!F || STL < F.from) return; const f0 = worldD(F.at || 0, 0) - SD + keyVal(F.front, STL);
  hzFill(hzBand(f0, f0 + 9000, F.z0, F.z1, 20, 18), '#7a5a3a'); hzFill(hzBand(f0, f0 + 9000, F.z0, F.z1, 20, 30), 'rgba(150,112,70,0.8)');
  for (let k = 0; k < 14; k++) { const z = lerp(F.z0, F.z1, hh(k, 3.1)), ff = f0 + 60 * Math.sin(t * 5 + k) - 40 * hh(k, 1.7), r = 120 * hzK(ff, 60, z) * (0.7 + hh(k, 2)); if (!hzVis(ff, z)) continue; const p = P3s(ff, 80 + 50 * hh(k, 4.2), z); hzPuff(p[0], p[1], Math.max(4, r), 'rgba(240,232,214,1)', 'rgba(210,196,170,0.9)', 'rgba(170,150,120,0)'); }
}

// the storm wall: a dark mass filling the sky, with a pale rain curtain under it and lightning glowing inside

  const bot = H * Math.min(0.66, 0.3 + 0.25 * k), g = ctx.createLinearGradient(0, 0, 0, bot); g.addColorStop(0, '#161c26'); g.addColorStop(0.6, '#2c3644'); g.addColorStop(1, '#4a5666');
  ctx.fillStyle = g; ctx.beginPath(); ctx.moveTo(0, 0); ctx.lineTo(W, 0); ctx.lineTo(W, bot);
